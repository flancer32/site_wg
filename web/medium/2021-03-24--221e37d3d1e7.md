# Структура JS-кода: Классы

Использование классов в ES6 хорошо задокументировано на [MDN](https://developer.mozilla.org/ru/docs/Web/JavaScript/Reference/Classes). В этом посте я описываю свой собственный подход к структурированию JS-кода с классами (первая — [про функции](https://alex-gusev.medium.com/%D1%81%D1%82%D1%80%D1%83%D0%BA%D1%82%D1%83%D1%80%D0%B0-js-%D0%BA%D0%BE%D0%B4%D0%B0-%D1%84%D1%83%D0%BD%D0%BA%D1%86%D0%B8%D0%B8-f4acfe63e5a0), третья — про [модули](https://alex-gusev.medium.com/%D1%81%D1%82%D1%80%D1%83%D0%BA%D1%82%D1%83%D1%80%D0%B0-js-%D0%BA%D0%BE%D0%B4%D0%B0-%D0%BC%D0%BE%D0%B4%D1%83%D0%BB%D0%B8-a6dd36033e69)).

В JS существуют [восемь основных типов данных](https://learn.javascript.ru/types):

*   undefined
*   null
*   number
*   string
*   boolean
*   bigint
*   symbol
*   object

Первые семь типов — это примитивы. Переменные этих типов содержат одно значение. Объекты — это композитные сущности. В JS всё, что не является примитивом, является объектом (в том числе и функции). Классы в JS, по сути, являются “[синтаксическим сахаром](https://ru.wikipedia.org/wiki/%D0%A1%D0%B8%D0%BD%D1%82%D0%B0%D0%BA%D1%81%D0%B8%D1%87%D0%B5%D1%81%D0%BA%D0%B8%D0%B9_%D1%81%D0%B0%D1%85%D0%B0%D1%80)” и позволяют более привычным для [ООП-программистов](https://ru.wikipedia.org/wiki/%D0%9E%D0%B1%D1%8A%D0%B5%D0%BA%D1%82%D0%BD%D0%BE-%D0%BE%D1%80%D0%B8%D0%B5%D0%BD%D1%82%D0%B8%D1%80%D0%BE%D0%B2%D0%B0%D0%BD%D0%BD%D0%BE%D0%B5_%D0%BF%D1%80%D0%BE%D0%B3%D1%80%D0%B0%D0%BC%D0%BC%D0%B8%D1%80%D0%BE%D0%B2%D0%B0%D0%BD%D0%B8%D0%B5) образом конструировать объекты в JavaScript’е.

## Конструктор-функция

Создание объектов при помощи функций в JS до версии ES2015 (ES6) выглядело так:

// object constructor

function FunctionClass({name, age}) {

 // object properties

 this.name = name; // internal variables

 const me = this; // pin 'this' to use with internal functions // internal functions

 function fnInternal() {

 // use 'me' here and 'age' arg from constructor's scope

 // 'height' is a virtual property with accessors (see below)

 return `${me.name}: ${age} / ${me.height}`;

 } // object methods

 this.fnInstance = function () {

 // we can use 'this', internal vars & funcs

 console.log(`${fnInternal()}.`);

 }

}FunctionClass.prototype.fnProto = function () {

 // we can use 'this' but cannot use

 // constructor's vars (me, age) & funcs (fnInternal)

 console.log(`${this.name}.`);

}// add 'height' property with accessors

Object.defineProperty(FunctionClass.prototype, 'height', {

 get: function () {

 return this._height;

 },

 set: function (value) {

 this._height = value;

 }

});FunctionClass.fnStatic = function () {

 // we can't use object's data here

 // 'this' points to 'FunctionClass' itself

 console.log(`Name: ${this.name}`);

};// create new instance

const objF = new FunctionClass({name: 'alex', age: 48});

objF.height = 175;

objF.fnInstance();

objF.fnProto();

FunctionClass.fnStatic();
## Использование классов

Аналогичный по функциональности код с использованием возможностей ES2015+ (ES6+):

class RegularClass {

 name;

 #height; constructor({name, age}) {

 // object properties

 this.name = name; // internal variables

 const me = this; // pin 'this' to use with internal funcs // internal functions

 function fnInternal() {

 // use 'me' here and 'age' arg from constructor's scope

 return `${me.name}: ${age} / ${me.#height}`;

 } // object methods

 this.fnInstance = function () {

 // we can use 'this', internal vars & funcs

 console.log(`${fnInternal()}.`);

 }

 } // add 'height' property with accessors

 get height() { return this.#height; }

 set height(val) { this.#height = val; } // add function to prototype

 fnProto() {

 // we can use 'this' but cannot use

 // constructor's vars (me, age) & funcs (fnInternal)

 console.log(`${this.name}.`);

 } static fnStatic() {

 // we can't use object's data here

 // 'this' points to 'FunctionClass' itself

 console.log(`Name: ${this.name}`);

 }

}// create new instance

const objC = new RegularClass({name: 'alex', age: 48});

objC.height = 175;

objC.fnInstance();

objC.fnProto();

RegularClass.fnStatic();
## Секции кода

Можно выделить следующие части при объявлении класса:

*   объявление свойств объекта;
*   конструктор с объявлением методов экземпляра объекта;
*   объявление методов прототипа объекта;
*   статические методы класса;

class RegularClass {

 // DEFINE PROPS

 static statProp = 4;

 pubProp = 8;

 #privProp = 16;

 get accessProp() {return this.#privProp};

 set accessProp(val) {this.#privProp = val}; // CONSTRUCTOR

 constructor(spec) {

 // PARSE INPUT, INIT PROPS, DEFINE WORKING VARS

 this.pubProp = spec;

 const me = this; // DEFINE INNER FUNCTIONS

 function fnInternal() {} // DEFINE INSTANCE METHODS

 this.fnInstance = function () { }

 } // DEFINE PROTO METHODS

 fnProto() {} // DEFINE STATIC METHODS

 static fnStatic() {}

}
Новые возможности JS по работе с классами доступны [не везде](https://developer.mozilla.org/en-US/docs/Web/JavaScript/Reference/Classes#browser_compatibility) (в частности, в Safari), поэтому приватные свойства лучше не использовать, а статические методы лучше делать по-старинке:

class RegularClass { }// DEFINE STATIC PROPS & METHODS

RegularClass.statProp = 2;

RegularClass.fnStatic = function () { }
Поднятие (hoisting) не работает для определений классов, поэтому сначала определяем класс, а потом его используем.

Секции кода без самого кода:

class RegularClass {

 // DEFINE PROPS // CONSTRUCTOR

 constructor(spec) {

 // PARSE INPUT, INIT PROPS, DEFINE WORKING VARS // DEFINE INNER FUNCTIONS // DEFINE INSTANCE METHODS }

 // DEFINE PROTO METHODS // DEFINE STATIC METHODS

}
